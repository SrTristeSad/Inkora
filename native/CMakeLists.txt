cmake_minimum_required(VERSION 3.16)
project(InkoraCore VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(inkora_core SHARED inkora_core.cpp)
target_compile_definitions(inkora_core PRIVATE INKORA_CORE_BUILD)
target_include_directories(inkora_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
  target_compile_options(inkora_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(inkora_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(inkora_core_test tests/core_test.cpp)
  target_link_libraries(inkora_core_test PRIVATE inkora_core)
  add_test(NAME InkoraCorePaint COMMAND inkora_core_test)
endif()
